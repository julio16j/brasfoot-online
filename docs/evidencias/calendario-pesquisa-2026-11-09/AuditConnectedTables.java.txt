package brasfoot;
import java.io.*;import java.nio.file.*;import java.util.*;import java.awt.*;import java.lang.reflect.*;import javax.swing.*;import javax.imageio.ImageIO;
public class AuditConnectedTables {
 static Object call(Object target,String name,Class<?>[] types,Object...args)throws Exception{Method m=target.getClass().getDeclaredMethod(name,types);m.setAccessible(true);return m.invoke(target,args);}
 public static void main(String[] args)throws Exception{
  Path root=Paths.get(args[0]);System.setProperty("user.dir",root.resolve("game/Brasfoot22-23").toString());UIManager.setLookAndFeel("javax.swing.plaf.metal.MetalLookAndFeel");
  final NativeClient[] client={null};SwingUtilities.invokeAndWait(()->{client[0]=new NativeClient(root);((JFrame)Data.get(client[0],"frame")).setVisible(true);});
  call(client[0],"connect",new Class<?>[]{String.class,String.class},args[1],args[2]);
  long deadline=System.currentTimeMillis()+15000;while(Data.get(client[0],"latest")==null && System.currentTimeMillis()<deadline)Thread.sleep(50);
  if(Data.get(client[0],"latest")==null)throw new AssertionError("Connected screen never became ready");
  Map<String,Object> result=new LinkedHashMap<String,Object>();Map<String,Object> latest=Data.map(Data.get(client[0],"latest"));result.put("viewerCoachId",latest.get("viewerCoachId"));result.put("phase",Data.map(latest.get("round")).get("phase"));
  SwingUtilities.invokeAndWait(()->{
   javax.swing.Timer timer=new javax.swing.Timer(300,null);timer.addActionListener(event->{try{for(Window w:Window.getWindows())if(w instanceof JDialog && w.isShowing()){
    JDialog d=(JDialog)w;if(d.getContentPane().getComponentCount()==0 || !(d.getContentPane().getComponent(0) instanceof a.ay))continue;
    a.ay panel=(a.ay)d.getContentPane().getComponent(0);result.put("standingsRows",((JTable)Data.get(panel,"wp")).getRowCount());result.put("fixturesRows",((JTable)Data.get(panel,"vN")).getRowCount());result.put("calendarDates",c.a.SR.R().size());JComboBox combo=(JComboBox)Data.get(panel,"va");result.put("competitionCount",combo.getItemCount());java.util.List<String> names=new ArrayList<String>();for(int index=0;index<combo.getItemCount();index++){Object item=combo.getItemAt(index);names.add(item instanceof best.at?((best.at)item).getNome():String.valueOf(item));}result.put("competitions",names);
    java.awt.image.BufferedImage initial=new java.awt.image.BufferedImage(d.getWidth(),d.getHeight(),java.awt.image.BufferedImage.TYPE_INT_RGB);java.awt.Graphics2D initialGraphics=initial.createGraphics();d.printAll(initialGraphics);initialGraphics.dispose();ImageIO.write(initial,"png",root.resolve("connected-tables-statistics.png").toFile());
    try{JLabel rounds=(JLabel)Data.get(panel,"xH");rounds.dispatchEvent(new java.awt.event.MouseEvent(rounds,java.awt.event.MouseEvent.MOUSE_CLICKED,System.currentTimeMillis(),0,3,3,1,false,java.awt.event.MouseEvent.BUTTON1));result.put("roundsActionCompleted",true);}catch(Throwable error){result.put("roundsActionFailure",error.getClass().getName()+": "+error.getMessage());}
    result.put("fixturesRowsAfterRounds",((JTable)Data.get(panel,"vN")).getRowCount());result.put("roundsVisible",((JPanel)Data.get(panel,"yg")).isVisible());result.put("groupCount",((JComboBox)Data.get(panel,"xz")).getItemCount());
    java.awt.image.BufferedImage image=new java.awt.image.BufferedImage(d.getWidth(),d.getHeight(),java.awt.image.BufferedImage.TYPE_INT_RGB);java.awt.Graphics2D graphics=image.createGraphics();d.printAll(graphics);graphics.dispose();ImageIO.write(image,"png",root.resolve("connected-tables.png").toFile());result.put("captureType","Swing printAll of actual connected dialog");timer.stop();d.dispose();
   }}catch(Exception error){timer.stop();error.printStackTrace();for(Window w:Window.getWindows())if(w instanceof JDialog)w.dispose();}});
   timer.start();JPanel preparation=(JPanel)Data.get(client[0],"preparationPanel");((JButton)Data.get(preparation,"HS")).doClick();
  });
  SwingUtilities.invokeAndWait(()->{try{call(client[0],"close",new Class<?>[]{});}catch(Exception error){throw new RuntimeException(error);}});
  Files.write(root.resolve("connected-tables.json"),Json.write(result).getBytes("UTF-8"));System.out.println(Json.write(result));System.exit(0);
 }
}
