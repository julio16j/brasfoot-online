package brasfoot;
import java.io.*;import java.nio.file.*;import java.util.*;import java.awt.*;import java.lang.reflect.*;import javax.swing.*;import javax.imageio.ImageIO;
public class AuditConnectedSearch {
 static Object call(Object target,String name,Class<?>[] types,Object...args)throws Exception{Method m=target.getClass().getDeclaredMethod(name,types);m.setAccessible(true);return m.invoke(target,args);}
 public static void main(String[] args)throws Exception{
  Path root=Paths.get(args[0]);System.setProperty("user.dir",root.resolve("game/Brasfoot22-23").toString());UIManager.setLookAndFeel("javax.swing.plaf.metal.MetalLookAndFeel");
  final NativeClient[] client={null};SwingUtilities.invokeAndWait(()->{client[0]=new NativeClient(root);((JFrame)Data.get(client[0],"frame")).setVisible(true);});
  call(client[0],"connect",new Class<?>[]{String.class,String.class},args[1],args[2]);
  long deadline=System.currentTimeMillis()+15000;while(Data.get(client[0],"latest")==null && System.currentTimeMillis()<deadline)Thread.sleep(50);
  if(Data.get(client[0],"latest")==null)throw new AssertionError("Connected screen never became ready");
  Map<String,Object> result=new LinkedHashMap<String,Object>();Map<String,Object> latest=Data.map(Data.get(client[0],"latest"));result.put("viewerCoachId",latest.get("viewerCoachId"));result.put("phase",Data.map(latest.get("round")).get("phase"));result.put("query","Federico Chiesa");
  SwingUtilities.invokeAndWait(()->{
   javax.swing.Timer timer=new javax.swing.Timer(300,null);timer.addActionListener(event->{try{for(Window w:Window.getWindows())if(w instanceof JDialog && w.isShowing()){
    JDialog d=(JDialog)w;if(d.getContentPane().getComponentCount()==0 || !(d.getContentPane().getComponent(0) instanceof a.gx))continue;
    a.gx panel=(a.gx)d.getContentPane().getComponent(0);Object filters=Data.get(panel,"zi");((JTextField)Data.get(filters,"uf")).setText("Federico Chiesa");((JButton)Data.get(panel,"yY")).doClick();
    result.put("rows",((JTable)Data.get(panel,"zj")).getRowCount());result.put("dialogTitle",d.getTitle());result.put("nativePlayerIndex",c.a.SR.O().size());result.put("nativeClubIndex",c.a.SR.P().size());result.put("calendarDates",c.a.SR.R().size());
    java.util.List<String> listeners=new ArrayList<String>();for(java.awt.event.MouseListener listener:((JTable)Data.get(panel,"zj")).getMouseListeners())listeners.add(listener.getClass().getName());result.put("mouseListeners",listeners);
    java.awt.image.BufferedImage image=new java.awt.image.BufferedImage(d.getWidth(),d.getHeight(),java.awt.image.BufferedImage.TYPE_INT_RGB);java.awt.Graphics2D graphics=image.createGraphics();d.printAll(graphics);graphics.dispose();ImageIO.write(image,"png",root.resolve("connected-search.png").toFile());result.put("captureType","Swing printAll of actual connected dialog");timer.stop();d.dispose();
   }}catch(Exception error){timer.stop();error.printStackTrace();for(Window w:Window.getWindows())if(w instanceof JDialog)w.dispose();}});
   timer.start();JPanel preparation=(JPanel)Data.get(client[0],"preparationPanel");((JButton)Data.get(preparation,"yY")).doClick();
  });
  SwingUtilities.invokeAndWait(()->{try{call(client[0],"close",new Class<?>[]{});}catch(Exception error){throw new RuntimeException(error);}});
  Files.write(root.resolve("connected-search.json"),Json.write(result).getBytes("UTF-8"));System.out.println(Json.write(result));System.exit(0);
 }
}
